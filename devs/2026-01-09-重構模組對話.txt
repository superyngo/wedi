  4.3 使用範例

  use wedi_core::{EmbeddedEditor, EditorConfig};
  use ratatui::prelude::*;

  fn main() -> Result<()> {
      // 建立編輯器
      let mut editor = EmbeddedEditor::new()
          .with_content("Hello, World!");

      // 配置
      editor.config()
          .line_numbers(true)
          .wrap_mode(true)
          .theme("Solarized Dark");

      // 自訂按鍵
      editor.keymap()
          .bind(ctrl('s'), Command::Custom("save"))
          .bind(ctrl('q'), Command::Custom("quit"));

      // 主迴圈
      loop {
          terminal.draw(|f| {
              editor.render(f.area(), f.buffer_mut());
          })?;

          if let Event::Key(key) = event::read()? {
              match editor.handle_key(key) {
                  EditorEvent::QuitRequested => break,
                  EditorEvent::SaveRequested => save(&editor),
                  _ => {}
              }
          }
      }
      Ok(())
  }
